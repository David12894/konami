// Render Functions
function displayPlayers(players) {
    const grid = document.getElementById('player-grid');
    grid.innerHTML = players.map(p => `
        <div class="player-card">
            <span class="rating-badge">${p.rating}</span>
            <div class="player-pos">${p.position}</div>
            <div class="player-name">${p.name}</div>
            <div class="card-details">
                <div><strong>Playstyle:</strong> ${p.style}</div>
                <div><strong>Theme Pack:</strong> ${p.pack}</div>
            </div>
        </div>
    `).join('');
}

window.filterPlaystyle = (styleName) => {
    const data = tacticsData[styleName];
    const detailsContainer = document.getElementById('tactic-details');
    
    const buttons = document.querySelectorAll('.tactic-btn');
    buttons.forEach(b => {
        if(b.textContent.includes(styleName)) b.classList.add('active');
        else b.classList.remove('active');
    });

    detailsContainer.innerHTML = `
        <h4>⚙️ Layout: ${data.formation}</h4>
        <p>${data.description}</p>
    `;
};

// AUTOMATIC WEEKLY THURSDAY SERVER MAINTENANCE TIMING CALCULATOR
function updateMaintenanceCountdown() {
    const timerElement = document.getElementById('maintenance-timer');
    const statusTextElement = document.getElementById('maintenance-status-text');
    const now = new Date();
    
    // Create target time pointing to Thursday at 08:00 UTC
    let targetEnd = new Date();
    targetEnd.setUTCHours(8, 0, 0, 0);
    
    // Calculate days until next Thursday (Thursday day index is 4)
    let daysDiff = (4 - now.getUTCDay() + 7) % 7;
    
    if (daysDiff === 0) {
        // If it is Thursday but past 08:00 UTC, target next week's Thursday
        if (now.getUTCHours() >= 8) {
            daysDiff = 7;
        }
    }
    
    targetEnd.setUTCDate(now.getUTCDate() + daysDiff);
    
    // If current time is on Thursday between 02:00 UTC and 08:00 UTC, maintenance is live
    const isMaintenanceLive = (now.getUTCDay() === 4 && now.getUTCHours() >= 2 && now.getUTCHours() < 8);
    
    if (isMaintenanceLive) {
        statusTextElement.textContent = "⚠️ Server Maintenance Is LIVE! Servers Up In:";
        statusTextElement.style.color = "#ef4444"; // Warning Red
    } else {
        statusTextElement.textContent = "⚡ Next Scheduled Thursday Maintenance Finishes In:";
        statusTextElement.style.color = "#cbd5e1";
    }

    const msLeft = targetEnd - now;
    
    const hours = Math.floor(msLeft / (1000 * 60 * 60));
    const minutes = Math.floor((msLeft % (1000 * 60 * 60)) / (1000 * 60));
    const seconds = Math.floor((msLeft % (1000 * 60)) / 1000);
    
    // Format numeric values with leading zeros
    const hDisplay = String(hours).padStart(2, '0');
    const mDisplay = String(minutes).padStart(2, '0');
    const sDisplay = String(seconds).padStart(2, '0');
    
    timerElement.textContent = `${hDisplay}h ${mDisplay}m ${sDisplay}s`;
}

    const filtered = playerDatabase.filter(p => p.name.toLowerCase().includes(val) || p.position.toLowerCase().includes(val));
    displayPlayers(filtered);
});
